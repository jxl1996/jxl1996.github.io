面试题: 100个任务,5个任务并发执行

基础实现:

```go
package main

import (
	"fmt"
	"sync"
)


func task(id int) {
	fmt.Printf("task %d\n", id)
}
func main() {
	numTasks := 100
	concurrency := 5
	var wg sync.WaitGroup
	taskChan := make(chan int)

	for i := 0; i < concurrency; i++ {
		wg.Add(1)

		go func() {
			defer wg.Done()

			for taskId := range taskChan {
				task(taskId)
			}
		}()
	}

	for i := 0; i < numTasks; i++ {
		taskChan <- i
	}
	close(taskChan)
	wg.Wait()
}

```

> [!CAUTION]
>
> 为什么最后必须使用`close(taskChan)`关闭管道???
>
> 因为 worker goroutine 是通过 `for range taskChan` 来消费任务的，
>  **range 会一直阻塞读取 channel，直到 channel 被关闭**。
>
> 如果不调用 `close(taskChan)`：
>
> - 所有任务发送完后，没有新的数据
> - worker goroutine 会一直阻塞在 `range`
> - `wg.Done()` 无法执行
> - 主 goroutine 在 `wg.Wait()` 一直等
>
> 👉 最终导致 **所有 goroutine 都在等待 → 死锁**

---

简单封装:

```go
package main

import (
	"fmt"
	"sync"
)

type Pool struct {
	taskChan chan func()
	wg       sync.WaitGroup
}

// 创建协程池
func NewPool(workerNum int) *Pool {
	p := &Pool{
		taskChan: make(chan func()),
	}

	for i := 0; i < workerNum; i++ {
		p.wg.Add(1)
		go func() {
			defer p.wg.Done()

			for task := range p.taskChan {
				task()
			}
		}()
	}

	return p
}

// 提交任务
func (p *Pool) Submit(task func()) {
	p.taskChan <- task
}

// 关闭协程池
func (p *Pool) WaitAndClose() {
	close(p.taskChan)
	p.wg.Wait()
}

func main() {
	p := NewPool(5)

	for i := 0; i < 100; i++ {
		j := i // ⚠️ 防止闭包问题
		p.Submit(func() {
			fmt.Println("task:", j)
		})
	}

	p.WaitAndClose()
}

```
