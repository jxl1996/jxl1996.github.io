面试题: 两个 goroutine 交替打印奇偶数

## 1. 使用chan实现

基础思路:

```go
package main

import (
	"fmt"
	"sync"
)

func main() {
	oddChan := make(chan struct{})
	evenChan := make(chan struct{})

	wg := sync.WaitGroup{}
	wg.Add(2)

	// 打印奇数
	go func() {
		defer wg.Done()
		for i := 1; i <= 10; i += 2 {
			<-oddChan
			fmt.Println("奇数:", i)
			evenChan <- struct{}{}
		}
	}()

	// 打印偶数
	go func() {
		defer wg.Done()
		for i := 2; i <= 10; i += 2 {
			<-evenChan
			fmt.Println("偶数:", i)
			oddChan <- struct{}{}
		}
	}()

	// 启动奇数打印
	oddChan <- struct{}{}

	wg.Wait()
}

```

上面代码结果正确但是会报死锁错误:

```
奇数: 1
偶数: 2
奇数: 3
偶数: 4
奇数: 5
偶数: 6
奇数: 7
偶数: 8
奇数: 9
偶数: 10
fatal error: all goroutines are asleep - deadlock!
```

解决思路: 打印最后一个偶数的时候, 不再向oddChan发送信号

```go
// 打印偶数
go func() {
    defer wg.Done()
    for i := 2; i <= 10; i += 2 {
        <-evenChan
        fmt.Println("偶数:", i)
        if i == 10 {
            break
        }
        oddChan <- struct{}{}
    }
}()
```



更优化版本: 可以更加灵活控制从奇数还是偶数开始打印

```go
package main

import (
	"fmt"
	"sync"
)

func main() {
	oddChan := make(chan struct{})
	evenChan := make(chan struct{})
	doneChan := make(chan struct{}, 1)

	maxNum := 10
	wg := sync.WaitGroup{}
	wg.Add(2)

	// 打印奇数
	go func() {
		defer wg.Done()
		for i := 1; i <= maxNum; i += 2 {
			<-oddChan

			fmt.Println("奇数:", i)
			select {
			case <-doneChan:
				break
			case evenChan <- struct{}{}:
			}
		}
		doneChan <- struct{}{}
	}()

	// 打印偶数
	go func() {
		defer wg.Done()
		for i := 2; i <= maxNum; i += 2 {
			<-evenChan
			fmt.Println("偶数:", i)

			select {
			case <-doneChan:
				break
			case oddChan <- struct{}{}:
			}
		}
		doneChan <- struct{}{}
	}()

	// 启动奇数打印
	// oddChan <- struct{}{}
	// 启动偶数打印
	evenChan <- struct{}{}

	wg.Wait()
}

```

## 2. 使用 sync.Cond 实现

```go
package main

import (
	"fmt"
	"sync"
)

func main() {
	var mu sync.Mutex
	cond := sync.NewCond(&mu)
	flag := true // true 表示打印奇数，false 表示打印偶数
	var wg sync.WaitGroup
	wg.Add(2)
	maxNum := 10

	// 打印奇数
	go func() {
		defer wg.Done()
		for i := 1; i <= maxNum; i += 2 {
			mu.Lock()
			for !flag {
				cond.Wait()
			}

			fmt.Println("奇数:", i)
			flag = false
			cond.Signal()

			mu.Unlock()
		}

	}()

	// 打印偶数
	go func() {
		defer wg.Done()
		for i := 2; i <= maxNum; i += 2 {
			mu.Lock()
			for flag {
				cond.Wait()
			}
			fmt.Println("偶数:", i)
			flag = true
			cond.Signal()
			mu.Unlock()
		}

	}()

	wg.Wait()
}

```
